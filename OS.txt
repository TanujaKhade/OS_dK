Q1. Write a C program to simulate Banker's algorithm for the purpose of deadlock avoidance.
Consider the following snapshot of system, A, B, C and D is the resource type.

#include<stdio.h>

int n,m;
int alloc[10][10],max[10][10],need[10][10];
int available[10],total[10],work[10],finish[10],seq[10],req[10];
int pno;
void accept()
{
	int i,j;
	printf("Enter no of process:");
	scanf("%d",&n);
	printf("Enter the no of resource type:");
	scanf("%d",&m);
	printf("Enter total instances of each resource type\n");
	for(j=0;j<m;j++)
	{
		printf("%c:",65+j);
		scanf("%d",&total[j]);
	}
	printf("Enter the allocation\n");
	for(i=0;i<n;i++)
	{
		printf("p%d:\n",i);
		for(j=0;j<m;j++)
		{
			printf("%c:",65+j);
			scanf("%d",&alloc[i][j]);
		}
	}
	printf("Enter max need:\n");
	for(i=0;i<n;i++)
	{
		printf("p%d:\n",i);
		for(j=0;j<m;j++)
		{
			printf("%c:",65+j);
			scanf("%d",&max[i][j]);
		}
	}
}
void calc_avail()
{
	int i,j,s;
	for(j=0;j<m;j++)
	{
		s=0;
		for(i=0;i<n;i++)
			s=s+alloc[i][j];

		available[j]=total[j]-s;
	}
}
void calc_need()
{
	int i,j;
	for(i=0;i<n;i++)
	{
		for(j=0;j<m;j++)
		{
			need[i][j]=max[i][j]-alloc[i][j];
		}
	}
}
void print()
{
	int i,j;
	printf("\n\tAllocation\t\tMax\t\tNeed\n\t");
	for(i=0;i<m;i++)
	{
		printf("%3c",65+i);
	}
	printf("\t");
	for(i=0;i<m;i++)
	{
		printf("%3c",65+i);
	}
	printf("\t");
	for(i=0;i<m;i++)
	{
		printf("%3c",65+i);
	}
	for(i=0;i<n;i++)
	{
		printf("\np%d\t",i);
		for(j=0;j<m;j++)
			printf("%3d",alloc[i][j]);
		printf("\t");
		for(j=0;j<m;j++)
			printf("%3d",max[i][j]);
		printf("\t");
		for(j=0;j<m;j++)
			printf("%3d",need[i][j]);
	}
	printf("\n\nAvailable\n");
	for(j=0;j<m;j++)
	{
		printf("%3c\t",65+j);
	}
	printf("\n");
	for(j=0;j<m;j++)
	{
		printf("%3d\t",available[j]);
	}
	printf("\n");
}
int find()
{
	int i,j;

	for(i=0;i<n;i++)
	{
		if(!finish[i])
		{
			for(j=0;j<m;j++)
			{
				if(need[i][j]>work[j])
					break;
			}

			if(j==m)
				return i;
		}
	}

	return -1;
}
void bankers()
{
	int i,j,k=0;
	for(i=0;i<n;i++)
		finish[i]=0;
	for(j=0;j<m;j++)
		work[j]=available[j];
	while((i=find())!=-1)
	{
		printf("\nNeed%d(",i);
		for(j=0;j<m;j++)
			printf("%d,",need[i][j]);
		printf("\b)<=work(");
		for(j=0;j<m;j++)
			printf("%d,",work[j]);
		printf("\b)");
		finish[i]=1;
		for(j=0;j<m;j++)
			work[j]=work[j]+alloc[i][j];
		printf("\nwork(");
		for(j=0;j<m;j++)
			printf("%d,",work[j]);
		printf("\b)\nFinish(");
		for(j=0;j<n;j++)
			printf("%d,",finish[j]);
		printf("\b)");
		seq[k++]=i;
	}
	if(k==n)
	{
		printf("\n\nSystem is in safe state:\n");
		printf("Safe sequence: ");
		for(j=0;j<n;j++)
			printf("p%d-",seq[j]);
	}
	else
	{
		printf("\n\nSystem is in unsafe state");
	}
}
int main()
{
	int i,j;
	system("clear");
	accept();
	calc_avail();
	calc_need();
	print();
	bankers();
	printf("\n\nEnter process no:");
	scanf("%d",&pno);
	printf("Enter request of process p%d\n",pno);
	for(j=0;j<m;j++)
	{
		printf("%c:",65+j);
		scanf("%d",&req[j]);
	}
	for(j=0;j<m;j++)
	{
		if(req[j]>need[pno][j])
			break;
	}
	if(j==m)
	{
		for(j=0;j<m;j++)
		{
			if(req[j]>available[j])
				break;
		}
		if(j==m)
		{
			for(j=0;j<m;j++)
			{
				available[j]=available[j]-req[j];
				alloc[pno][j]=alloc[pno][j]+req[j];
				need[pno][j]=need[pno][j]-req[j];
			}
			print();

			bankers();
		}
		else
		{
			printf("\nProcess p%d has to wait",pno);
		}
	}
	else
	{
		printf("\nRequest of process p%d cannot be generated",pno);
	}
	return 0;
}


Q2. Write a simulation program for disk scheduling using SSTF algorithm. Accept total number
of disk blocks, disk request string, and current head position from the user. Display the list of
requests in the order in which it is served. Also display the total number of head movements.
Request string: 98, 183, 37, 122, 14, 124, 65
Start Head Position: 53



#include<stdio.h>
#include<stdlib.h>

int main()
{
    int blocks, n, i, j, head;
    int request[100], visited[100] = {0};
    int seek = 0, min, pos, diff;
    printf("Enter total number of disk blocks: ");
    scanf("%d", &blocks);
    printf("Enter number of disk requests: ");
    scanf("%d", &n);
    printf("Enter disk request string:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &request[i]);
    }
    printf("Enter current head position: ");
    scanf("%d", &head);
    printf("\nRequests served in order:\n");
    printf("%d", head);
    for(i = 0; i < n; i++)
    {
        min = 99999;
        pos = -1;
        for(j = 0; j < n; j++)
        {
            if(visited[j] == 0)
            {
                diff = abs(head - request[j]);
                if(diff < min)
                {
                    min = diff;
                    pos = j;
                }
            }
        }
        visited[pos] = 1;
        seek = seek + min;
        head = request[pos];
        printf(" -> %d", head);
    }
    printf("\n\nTotal number of head movements = %d\n", seek);
    return 0;
}

Q2. Write a simulation program for disk scheduling using SCAN algorithm. Accept total number
of disk blocks, disk request string, and current head position from the user. Display the list of
requests in the order in which it is served. Also display the total number of head movements.
Request string: 82, 170, 43, 140, 24, 16, 190, 65
Starting Head Position = 50
Direction: Left


#include<stdio.h>
#include<stdlib.h>

int main()
{
    int blocks,n,i,j,head;
    int request[100],visited[100]={0};
    int left[100],right[100];
    int l=0,r=0;
    int seek=0,temp;
    char direction;
    printf("Enter total number of disk blocks: ");
    scanf("%d",&blocks);
    printf("Enter number of disk requests: ");
    scanf("%d",&n);
    printf("Enter disk request string:\n");
    for(i=0;i<n;i++)
        scanf("%d",&request[i]);
    printf("Enter starting head position: ");
    scanf("%d",&head);
    printf("Enter direction (L/R): ");
    scanf(" %c",&direction);
    for(i=0;i<n;i++)
    {
        if(request[i]<head)
            left[l++]=request[i];
        else
            right[r++]=request[i];
    }
    for(i=0;i<l-1;i++)
    {
        for(j=i+1;j<l;j++)
        {
            if(left[i]<left[j])
            {
                temp=left[i];
                left[i]=left[j];
                left[j]=temp;
            }
        }
    }
    for(i=0;i<r-1;i++)
    {
        for(j=i+1;j<r;j++)
        {
            if(right[i]>right[j])
            {
                temp=right[i];
                right[i]=right[j];
                right[j]=temp;
            }
        }
    }
    printf("\nRequests served in order:\n");
    printf("%d",head);
    if(direction=='L' || direction=='l')
    {
        for(i=0;i<l;i++)
        {
            seek=seek+abs(head-left[i]);
            head=left[i];
            printf(" -> %d",head);
        }
        if(head!=0)
        {
            seek=seek+head;
            head=0;
            printf(" -> %d",head);
        }
        for(i=0;i<r;i++)
        {
            seek=seek+abs(head-right[i]);
            head=right[i];
            printf(" -> %d",head);
        }
    }
    else
    {
        for(i=0;i<r;i++)
        {
            seek=seek+abs(head-right[i]);
            head=right[i];
            printf(" -> %d",head);
        }
        if(head!=blocks-1)
        {
            seek=seek+abs(head-(blocks-1));
            head=blocks-1;
            printf(" -> %d",head);
        }
        for(i=0;i<l;i++)
        {
            seek=seek+abs(head-left[i]);
            head=left[i];
            printf(" -> %d",head);
        }
    }
    printf("\n\nTotal number of head movements = %d\n",seek);
    return 0;
}

Q1.  Write the simulation program for demand paging and show the page scheduling and total
number of page faults according to the FIFO page replacement algorithm. Assume the
memory of n frames.
Reference String: 3, 4, 5, 6, 3, 4, 7, 3, 4, 5, 6, 7, 2, 4, 6

#include<stdio.h>

int main()
{
    int n, i, j, k;
    int ref[100], frame[20];
    int pageFaults = 0;
    int pointer = 0;
    int found;
    printf("Enter number of frames: ");
    scanf("%d",&n);
    printf("Enter number of pages in reference string: ");
    scanf("%d",&k);
    printf("Enter reference string:\n");
    for(i=0;i<k;i++)
        scanf("%d",&ref[i]);
    for(i=0;i<n;i++)
        frame[i]=-1;
    printf("\nPage\t");
    for(i=0;i<n;i++)
        printf("F%d\t",i+1);
    printf("Status\n");
    for(i=0;i<k;i++)
    {
        found=0;
        for(j=0;j<n;j++)
        {
            if(frame[j]==ref[i])
            {
                found=1;
                break;
            }
        }
        if(found==0)
        {
            frame[pointer]=ref[i];
            pointer=(pointer+1)%n;
            pageFaults++;
        }
        printf("%d\t",ref[i]);
        for(j=0;j<n;j++)
        {
            if(frame[j]==-1)
                printf("-\t");
            else
                printf("%d\t",frame[j]);
        }
        if(found==0)
            printf("Page Fault");
        else
            printf("Hit");
        printf("\n");
    }
    printf("\nTotal number of page faults = %d\n",pageFaults);
    return 0;
}

        printf("\n");
    }

}

    request_resource();
    return 0;

CPU burst for different n number of processes should be input to the algorithm. Also give the

#include<stdio.h>
#include<stdlib.h>
#include<time.h>

{

Q2.  Implement the C program to accept n integers to be sorted. Main function creates child
process using fork system call. Parent process sorts the integers using bubble sort and waits
for child process using wait system call. Child process sorts the integers using insertion sort.


#include<stdio.h>
#include<unistd.h>
#include<sys/types.h>
#include<sys/wait.h>
#include<stdlib.h>

void bubble_sort(int a[],int n)
{
	int i,j,temp;

	for(i=0;i<n-1;i++)
	{
		for(j=0;j<n-i-1;j++)
		{
			if(a[j]>a[j+1])
			{
				temp=a[j];
				a[j]=a[j+1];
				a[j+1]=temp;
			}
		}
	}
}

void insertion_sort(int a[],int n)
{
	int i,j,key;

	for(i=1;i<n;i++)
	{
		key=a[i];
		j=i-1;

		while(j>=0 && a[j]>key)
		{
			a[j+1]=a[j];
			j--;
		}

		a[j+1]=key;
	}
}

void display(int a[],int n)
{
	int i;

	for(i=0;i<n;i++)
		printf("%d ",a[i]);

	printf("\n");
}

int main()
{

	int a[100],b[100];
	pid_t pid;

	printf("Enter number of integers: ");
	scanf("%d",&n);

	printf("Enter %d integers:\n",n);

	for(i=0;i<n;i++)
	{
		scanf("%d",&a[i]);
		b[i]=a[i];
	}

	{
		printf("Fork failed\n");
		return 1;
	}

	if(pid==0)
	{
		printf("\nChild Process\n");
		printf("Child Process ID: %d\n",getpid());

		insertion_sort(b,n);

		printf("Sorted array using Insertion Sort:\n");
		display(b,n);

		exit(0);
	}
	else
	{
		printf("\nParent Process\n");
		printf("Parent Process ID: %d\n",getpid());

		bubble_sort(a,n);

		printf("Sorted array using Bubble Sort:\n");
		display(a,n);

		wait(NULL);

		printf("Parent process completed after waiting for child.\n");
	}

	return 0;
}

	int ct;
	int tat;
	int wt;
	int done;
};
int main()
{
	struct process p[20];
	int n,i,time=0,count=0,pos;
	float avgwt=0,avgtat=0;
	srand(time(0));
	printf("Enter number of processes: ");
	scanf("%d",&n);
	for(i=0;i<n;i++)
	{
		p[i].pid=i+1;
		printf("\nEnter arrival time for P%d: ",i+1);
		scanf("%d",&p[i].at);
		printf("Enter first CPU burst for P%d: ",i+1);
		scanf("%d",&p[i].bt);
		p[i].done=0;
	}
	printf("\nGantt Chart:\n");
	while(count<n)
	{
		pos=-1;
		for(i=0;i<n;i++)
		{
			if(p[i].done==0 && p[i].at<=time)
			{
				if(pos==-1 || p[i].bt<p[pos].bt)
					pos=i;
			}
		}
		if(pos==-1)
		{
			time++;
			continue;
		}
		printf("| P%d ",p[pos].pid);
		time=time+p[pos].bt;
		p[pos].ct=time;
		p[pos].tat=p[pos].ct-p[pos].at;
		p[pos].wt=p[pos].tat-p[pos].bt;
		p[pos].done=1;
		avgwt=avgwt+p[pos].wt;
		avgtat=avgtat+p[pos].tat;
		count++;
	}
	printf("|\n");
	printf("\nProcess\tArrival\tCPU Burst\tCompletion\tTurnaround\tWaiting\n");
	for(i=0;i<n;i++)
	{
		printf("P%d\t%d\t%d\t\t%d\t\t%d\t\t%d\n",
		p[i].pid,p[i].at,p[i].bt,p[i].ct,p[i].tat,p[i].wt);
	}
	printf("\nAverage Waiting Time = %.2f",avgwt/n);
	printf("\nAverage Turnaround Time = %.2f",avgtat/n);
	printf("\n\nNext CPU Burst generated randomly:\n");
	for(i=0;i<n;i++)
	{
		printf("P%d = %d\n",p[i].pid,(rand()%10)+1);
	}
	printf("\n");
	return 0;
}

#include<sys/types.h>

#include<stdio.h>

#include<time.h>

struct process
{
	int pid;
	int at;
	int bt;
	int rt;
	int priority;
	int ct;
	int tat;
	int wt;
};
int main()
{
	struct process p[20];

Q2.  Write a program that demonstrates the use of nice() system call. After a child process is
started using fork(), assign higher priority to the child using nice() system call.


#include<stdio.h>
#include<unistd.h>
#include<sys/types.h>
#include<sys/wait.h>

int main()
{
	pid_t pid;
	int n;
	printf("Parent Process ID: %d\n",getpid());
	printf("Parent Nice Value: %d\n",nice(0));
	pid=fork();
	if(pid<0)
	{
		printf("Fork failed\n");
		return 1;
	}
	if(pid==0)
	{
		printf("\nChild Process\n");
		printf("Child Process ID: %d\n",getpid());
		printf("Nice Value Before: %d\n",nice(0));
		n=nice(-5);
		if(n==-1)
		{
			perror("nice");
			return 1;
		}
		printf("Nice Value After: %d\n",nice(0));
		printf("Child process has higher priority\n");
	}
	else
	{
		wait(NULL);
		printf("\nParent process completed\n");
	}
	return 0;
}

Q2.  Write the simulation program for demand paging and show the page scheduling and total
number of page faults according to the FIFO page replacement algorithm. Assume the
memory of n frames.
Reference String: 3, 4, 5, 6, 3, 4, 7, 3, 4, 5, 6, 7, 2, 4, 6


#include<stdio.h>

int main()
{
	int n,k,i,j;
	int ref[100],frame[20];
	int page,fault=0,found;
	int pos=0;
	printf("Enter number of frames: ");
	scanf("%d",&n);
	printf("Enter number of pages in reference string: ");
	scanf("%d",&k);
	printf("Enter reference string:\n");
	for(i=0;i<k;i++)
		scanf("%d",&ref[i]);
	for(i=0;i<n;i++)
		frame[i]=-1;
	printf("\nPage\t");
	for(i=0;i<n;i++)
		printf("F%d\t",i+1);
	printf("Status\n");
	for(i=0;i<k;i++)
	{
		page=ref[i];
		found=0;
		for(j=0;j<n;j++)
		{
			if(frame[j]==page)
			{
				found=1;
				break;
			}
		}
		if(found==0)
		{
			frame[pos]=page;
			pos=(pos+1)%n;
			fault++;
		}
		printf("%d\t",page);
		for(j=0;j<n;j++)
		{
			if(frame[j]==-1)
				printf("-\t");
			else
				printf("%d\t",frame[j]);
		}
		if(found==0)
			printf("Page Fault");
		else
			printf("Hit");
		printf("\n");
	}
	printf("\nTotal number of page faults = %d\n",fault);
	return 0;
}
